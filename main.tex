\section{Model}
\label{sec:model}

We analyze a platform model in which a human creator and an AI content generator compete for consumer views. The interaction is a two-stage game: in Stage~1, the platform sets its strategic parameters; in Stage~2, the human creator chooses an effort level. We solve the game by backward induction.

\subsection{Content Generation and Quality}

The human creator exerts effort $q_h \ge 0$, representing the quality of human-generated content, at a quadratic cost $C(q_h) = \tfrac{1}{2}q_h^2$. The AI generates content quality
\begin{equation}
    q_A = \alpha q_h + Q,
    \label{eq:qA}
\end{equation}
where $\alpha \in [0,1]$ is the AI's learning efficiency and $Q \ge 0$ is its baseline quality.

\subsection{Consumer Demand}

Consumers are uniformly distributed on $[0,1]$, with location $x$ indexing preference proximity to the AI (at~0) relative to the human (at~1). The utility a consumer at $x$ derives from each creator is
\begin{align}
    u_A &= q_A - t\,x\,\bigl(\beta_h\delta + (1-\delta)\bigr), \label{eq:uA}\\
    u_h &= q_h - t\,(1-x)\,(1-\beta_h\delta), \label{eq:uh}
\end{align}
where $t > 0$ is a mismatch cost parameter, $\beta_h \in [0,1]$ is the platform's promotion weight for the human creator, and $\delta \in (0,1)$ scales the algorithm's influence on mismatch costs. To ease notation, define
\begin{equation}
    m_A(\beta_h) \;\equiv\; \beta_h\delta + (1-\delta) \;=\; 1 - \delta(1-\beta_h), \qquad
    m_H(\beta_h) \;\equiv\; 1 - \beta_h\delta,
    \label{eq:mismatch}
\end{equation}
so that $u_A = q_A - tx\,m_A$ and $u_h = q_h - t(1-x)\,m_H$.

The specification has two key properties. First, both mismatch weights are strictly bounded away from zero for all $\beta_h \in [0,1]$: $m_A \in [1-\delta,\, 1]$ and $m_H \in [1-\delta,\, 1]$, with the floor $1-\delta > 0$ reflecting an irreducible taste-based friction that the algorithm cannot eliminate. Second, $\beta_h$ acts asymmetrically: increasing $\beta_h$ raises $m_A$ (penalizing the AI) while decreasing $m_H$ (rewarding the human). At the corners: as $\beta_h \to 0$, $m_A \to 1-\delta$ and $m_H \to 1$; as $\beta_h \to 1$, $m_A \to 1$ and $m_H \to 1-\delta$.

Assuming a partially covered market with outside option $u_0 > 0$ and setting $u_i = u_0$, the demand functions are
\begin{equation}
    D_A = \frac{q_A - u_0}{t\,m_A}, \qquad
    D_h = \frac{q_h - u_0}{t\,m_H}.
    \label{eq:demands}
\end{equation}

\subsection{Platform and Creator Payoffs}

The platform chooses the promotion weight $\beta_h \in [0,1]$ and compensation ratio $r \in [0,1]$ to maximize :
\begin{equation}
    u_p = \text{Revenue} - r\,D_h.
    \label{eq:up}
\end{equation}
The human creator's payoff is revenue minus effort cost:
\begin{equation}
    u_c = r\,D_h - \tfrac{1}{2}q_h^2.
    \label{eq:uc}
\end{equation}

    u_p = D_A + (1-r)\,D_h.
    \label{eq:up}
\end{equation}

\subsection{Stage 2: Human Creator's Optimal Effort}

\begin{proposition}[Optimal Human Effort]
\label{prop:qh}
The human creator's optimal effort and the AI's induced quality are
\begin{equation}
    q_h^* = \frac{r}{t\,m_H}, \qquad
    q_A^* = \frac{\alpha r}{t\,m_H} + Q.
    \label{eq:qhstar}
\end{equation}
\end{proposition}

\subsection{Stage 1: Platform's Optimal Compensation Rate}

\begin{proposition}[Optimal Compensation Rate]
\label{prop:rstar}
The platform's optimal compensation rate is
\begin{equation}
    r^* = \frac{1}{2}\left[1 + \frac{\alpha\,m_H}{m_A} + t\,m_H\,u_0\right]
        = \frac{1}{2}\left[1 + \frac{\alpha(1-\beta_h\delta)}{1-\delta(1-\beta_h)} + t(1-\beta_h\delta)\,u_0\right].
    \label{eq:rstar}
\end{equation}
\end{proposition}


\subsection{Stage 1: Platform's Optimal Promotion Weight}


\begin{assumption}\label{ass:params}
$\alpha \in (0,1]$, $\delta \in (0,1)$, $t > 0$, $u_0 > 0$, $Q \ge 0$, and for $\beta_h = \beta_h^*$
\[
\frac{\alpha(1-\beta_h\delta)}{1-\delta(1-\beta_h)} + t(1-\beta_h\delta)\,u_0 < 1.
\]
The following condition ensure the above holds for all $\beta_h \in [0,1]$:
\[
 t\,u_0 < 1 - \frac{\alpha}{1-\delta}
\]
This bounds the AI learning spillover, outside option pressure, and algorithmic influence to be collectively moderate enough that $r^*(\beta_h) \in (0,1)$ and $q_h^* > u_0$ for all $\beta_h \in [0,1]$, ensuring the platform retains a positive margin and the human creator participates.
\end{assumption}

\begin{proposition}[Optimal Promotion Weight]\label{prop:combined}
Under Assumption~\ref{ass:params}, the platform's optimal promotion weight is determined as follows:
\begin{enumerate}
\item[\textup{(i)}] If $Q \ge u_0$, the optimum is a corner solution: $\beta_h^* \in \{0,1\}$, chosen by comparing $u_p(0)$ and $u_p(1)$.
\item[\textup{(ii)}] If $Q < u_0$, define the survival threshold $\tilde\beta \in (0,1)$ as the unique solution to $q_A^*(\tilde\beta) = u_0$. Then either an interior optimum $\beta_h^* \in (\tilde\beta, 1)$ exists, or $\beta_h^* = 1$.
\end{enumerate}
\end{proposition}

\section{Equilibrium Analysis (Engagement-based Revenue)}
\label{sec:analysis}
We now transition to a platform model where revenue is driven by engagement quality rather than pure view counts.

The platform's utility function is:
\begin{equation}
u_{p} = q_{A}D_{A} + q_{h}D_{h} - r D_{h} = q_{A}D_{A} + (q_{h} - r)D_{h}
\end{equation}

\subsection{Stage 2: Human Creator's Optimal Effort}

Since the compensation structure for the human creator remains a flat rate $r$ per view, the creator's payoff function $u_{c} = r D_{h} - \frac{1}{2}q_{h}^{2}$ is unchanged. Consequently, the optimal effort choices are identical to those derived in Proposition \ref{prop:qh}:
\begin{equation}
q_{h}^{*} = \frac{r}{tm_{H}}, \quad q_{A}^{*} = \frac{\alpha r}{tm_{H}} + Q
\end{equation}

\subsection{Stage 1: Platform's Optimal Compensation Rate}
\begin{assumption}\label{ass:engagement}
$\alpha \in (0,1]$, $\delta \in (0,1)$, $t > 0$, $u_0 > 0$, $Q \ge 0$, and at the equilibrium $\beta_h^*$:
\begin{enumerate}
\item[\textup{(a)}] $\alpha^2 m_H + m_A(1 - tm_H) < 0$.
 
This is the second-order condition (concavity of $u_p$ in $r$). It requires $tm_H > 1$. Since $m_H$ is minimized at $\beta_h = 1$ where $m_H = 1-\delta$, a sufficient condition for all $\beta_h$ is
\[
t > \frac{1}{1-\delta} + \alpha^2.
\]
 
\item[\textup{(b)}] $u_0\,m_A(1 - tm_H) < \alpha\,m_H(2Q - u_0)$.
 
This ensures $r^* > 0$ (both numerator and denominator of $r^*$ are negative). If $2Q \ge u_0$, condition \textup{(b)} is automatic since the left-hand side is negative and the right-hand side is nonnegative. If $2Q < u_0$, the condition is equivalently
\[
u_0\,m_A(tm_H - 1) > \alpha\,m_H(u_0 - 2Q).
\]
\end{enumerate}
\end{assumption}

\begin{proposition}[Optimal Compensation Rate under Engagement]
\label{prop:r(eng)}
The platform's optimal engagement-based compensation rate is:
\begin{equation}
r^{*} = \frac{tm_{H} [ u_{0}m_{A}(1 - tm_{H}) - \alpha m_{H}(2Q - u_{0}) ]}{2 [ \alpha^{2}m_{H} + m_{A}(1 - tm_{H}) ]}
\end{equation}
\end{proposition}



\subsection{Stage 1: Platform's Optimal Promotion Weight}

\begin{assumption}\label{ass:eng-promotion}
Assumptions~\ref{ass:engagement} hold, and the platform has an initial incentive to promote the human creator: $du_p/d\beta_h\big|_{\beta_h=0} > 0$.
\end{assumption}
 

\begin{proposition}[Optimal Promotion Weight]
\label{prop:beta(eng)}
Under Assumption~\ref{ass:eng-promotion},  the optimal promotion weight $\beta_{h}^{*}$ falls into one of two regimes based on the AI's baseline quality $Q$:
\begin{itemize}
    \item[(i)] \textbf{Interior Solution (High Baseline Quality):} If $Q$ is sufficiently high, there exists a unique interior optimum $\beta_{h}^{*} \in (0, 1)$. The platform balances the marginal engagement gains driven by higher content quality against the marginal engagement losses from worsening the AI's mismatch friction.
    \item[(ii)] \textbf{Corner Solution (Low Baseline Quality):} If $Q$ is sufficiently small, the marginal quality gains globally dominate the AI mismatch penalty. The platform sets $\beta_{h}^{*} = 1$, fully prioritizing the human creator to maximize effort extraction for AI learning.
\end{itemize}
\end{proposition}


\section{Numerical Illustrations}
 
Our analytical results establish that the platform's optimal promotion and compensation policies depend qualitatively on the revenue model (view-based vs.\ engagement-based) and quantitatively on the AI's baseline quality~$Q$, learning efficiency~$\alpha$, and the mismatch environment~$(t, \delta)$. In this section, we use numerical experiments to illustrate these dependencies and draw comparisons across the two regimes.
 
Throughout, we fix a common set of primitive parameters $(\alpha, \delta, u_0)$ and select $t$ separately for each model to satisfy the respective assumptions. The view-based model uses $\alpha = 0.3$, $\delta = 0.5$, $u_0 = 0.2$, and $t = 0.8$ (satisfying Assumption~1: $\alpha/(1-\delta) + tu_0 = 0.76 < 1$). The engagement-based model shares $\alpha$, $\delta$, and $u_0$ but sets $t = 2.5$ (satisfying Assumption~2(a): $t > 1/(1-\delta) + \alpha^2 = 2.09$). Each experiment varies one parameter while holding the others at these baseline values.
 
\subsection{Promotion Policy Across Revenue Models}
 
We first plot the platform's optimal promotion weight $\beta_h^*$ against the AI's baseline quality $Q$ under both revenue models.
 
This comparison is the central question of the paper: how should a platform allocate visibility between human and AI content, and how does this depend on the platform's revenue structure? Our theory predicts qualitatively different behavior. Under view-based revenue, Proposition~3 shows that the platform chooses a corner solution when $Q \geq u_0$ and may admit an interior optimum only when $Q < u_0$. Under engagement-based revenue, Proposition~5 predicts a smooth transition: for low $Q$, the platform fully promotes the human creator ($\beta_h^* = 1$) to extract effort for AI learning, while for high $Q$, an interior solution emerges as the AI's friction drag grows quadratically in $Q$.
 
The numerical results should reveal three features: (i)~the discontinuous, bang-bang character of view-based promotion versus the smoother engagement-based policy; (ii)~the threshold value of $Q$ at which each model transitions between regimes; and (iii)~the direction of the shift---whether stronger AI baseline quality leads to more or less human promotion.
 
\subsection{Compensation Policy Across Revenue Models}
 
We next compare the optimal compensation rate $r^*$ as a function of $Q$ under both models.
 
Compensation is the platform's primary lever for incentivising human effort, and the two models produce structurally different formulas: view-based compensation (Proposition~2) always exceeds $1/2$ and depends on $Q$ only indirectly through $\beta_h^*$, while engagement-based compensation (Proposition~4) can be substantially lower and responds directly to $Q$ through the term $\alpha m_H(2Q - u_0)$ in the numerator. Plotting $r^*$ against $Q$ reveals how the platform's willingness to pay for human content varies with AI capability. In the view-based model, we expect $r^*$ to be relatively stable because the platform's revenue is linear in demand and the marginal value of an additional view is constant. In the engagement-based model, as $Q$ rises, the AI generates high-quality content autonomously, reducing the platform's marginal return to human effort and depressing $r^*$.
 
\subsection{Human Creator Welfare}
 
We plot the human creator's equilibrium payoff $u_c^* = r^* D_h^* - \frac{1}{2}(q_h^*)^2$ against $Q$.
 
Creator welfare captures the net benefit to the human participant after accounting for effort costs. This is policy-relevant: if platforms transition from view-based to engagement-based revenue models, it matters whether human creators are better or worse off. Our theory does not deliver a clean ranking between the two models, making numerical comparison essential. The key tension is that the engagement model demands higher mismatch costs ($t$ large), which in isolation would reduce demand, but the quality weighting of engagement amplifies the returns to high-quality content. The plot should clarify which effect dominates and how creator welfare responds to improvements in AI baseline quality---whether $Q$ acts as a complement (lifting all boats through learning spillovers) or a substitute (crowding out human effort and pay).
 
\subsection{Consumer Surplus}
 
We compute total consumer surplus and plot it against $Q$ under both models. For the partially covered market, consumer surplus from creator $i \in \{A, h\}$ is
\[
CS_i = \int_0^{D_i} (q_i - u_0 - t\,x\,m_i)\,dx = \frac{(q_i - u_0)^2}{2\,t\,m_i},
\]
and total surplus is $CS = CS_A + CS_h$.
 
Consumer welfare is the standard normative benchmark and the metric most relevant to regulators considering platform intervention. The two revenue models induce different quality--variety tradeoffs: view-based platforms maximise eyeballs and may tolerate lower quality, while engagement-based platforms internalise quality directly. As $Q$ increases, AI content improves exogenously, but the indirect effect on human content quality (through $\alpha$ and the equilibrium $r^*$, $\beta_h^*$) can go either way. The consumer surplus comparison reveals whether engagement-based revenue design, which aligns platform incentives more closely with content quality, translates into measurable welfare gains for consumers.
 
\subsection{Comparative Statics in Learning Efficiency}
 
We vary $\alpha$ and plot $\beta_h^*$, $r^*$, and $u_c^*$ for each model.
 
The AI's learning efficiency $\alpha$ governs the strength of the spillover from human effort to AI quality. In the view-based model, higher $\alpha$ increases the platform's return to human promotion (through the effort spillover term in equation~(12)), but also raises $r^*$ toward its upper bound, tightening Assumption~1. In the engagement-based model, $\alpha$ amplifies both the quality boost and the friction drag in the marginal promotion condition (equation~(28)), with ambiguous net effect. This experiment reveals whether the two models respond monotonically to $\alpha$ or whether non-monotonicities arise---for instance, whether there exists a level of learning efficiency beyond which the platform switches from promoting the human creator to suppressing promotion.
 
\subsection{Comparative Statics in Algorithmic Influence}
 
We vary $\delta$ and plot $\beta_h^*$, $r^*$, and $u_c^*$ for each model.
 
The parameter $\delta$ controls the platform's ability to steer consumers through algorithmic promotion. A larger $\delta$ makes promotion more effective (lowering the floor $1 - \delta$ on mismatch costs) but also makes the competitive landscape more lopsided. This experiment isolates the value of algorithmic power to the platform: does a more influential algorithm benefit the platform by sharpening its promotion tool, or does it destabilise the equilibrium by making the human creator's participation constraint harder to satisfy? Comparing across models reveals whether engagement-based platforms exploit algorithmic influence differently than view-based ones.

\noindent\textbf{Proof of Proposition~\ref{prop:qh}}
\label{app:proof_qh}
\begin{proof}
The human creator maximizes $u_c = \tfrac{r(q_h - u_0)}{t\,m_H} - \tfrac{1}{2}q_h^2$. The first-order condition $\tfrac{\partial u_c}{\partial q_h} = \tfrac{r}{t\,m_H} - q_h = 0$ gives $q_h^* = \tfrac{r}{t\,m_H}$. The second-order condition $\tfrac{\partial^2 u_c}{\partial q_h^2} = -1 < 0$ confirms a global maximum. Substituting into \eqref{eq:qA} yields $q_A^* = \tfrac{\alpha r}{t\,m_H} + Q$.\hfill$\blacksquare$
\end{proof}

\noindent\textbf{Proof of Proposition~\ref{prop:rstar}}
\label{app:proof_rstar}
\begin{proof}
Substituting $q_h^*$ and $q_A^*$ into \eqref{eq:up} and differentiating with respect to $r$:
\begin{equation}
    \frac{\partial u_p}{\partial r} = \frac{\partial D_A}{\partial r} + (1-r)\frac{\partial D_h}{\partial r} - D_h = 0.
\end{equation}
The relevant partial derivatives are
\begin{equation}
    \frac{\partial D_A}{\partial r} = \frac{\alpha}{t^2 m_A\,m_H}, \qquad
    \frac{\partial D_h}{\partial r} = \frac{1}{t^2 m_H^2}.
\end{equation}
Substituting and multiplying through by $t^2 m_H^2$:
\[
    \frac{\alpha\,m_H}{m_A} + (1-r) - r + t\,m_H\,u_0 = 0.
\]
Solving for $r$ gives \eqref{eq:rstar}. Note that the second derivative is -2, which shows $u_p$ is concave in $r$.
\hfill$\blacksquare$
\end{proof}



\noindent\textbf{Proof of Proposition~\ref{prop:combined}}
\begin{proof}
Throughout, write $m_A = 1 - \delta(1-\beta_h)$, $m_H = 1 - \beta_h\delta$, and define $y_A = 1/m_A$, $y_H = 1/m_H$, which are strictly convex since $y_A'' = 2\delta^2/m_A^3 > 0$ and $y_H'' = 2\delta^2/m_H^3 > 0$.
 
\medskip\noindent\textbf{Case (i): $Q \ge u_0$.} We first substitute the equilibrium qualities $q_h^* = \frac{r}{tm_H}$ and $q_A^* = \frac{\alpha r}{tm_H} + Q$ into the demand functions:
\begin{align*}
D_A &= \frac{\alpha r}{t^2 m_A m_H} + \frac{Q-u_0}{t m_A} \\
D_h &= \frac{r}{t^2 m_H^2} - \frac{u_0}{t m_H}
\end{align*}
Substituting these into $u_p$ and grouping by $r$ reveals a quadratic function in $r$:
\begin{equation}
u_p(r) = -B r^2 + A r + C
\end{equation}
where the coefficients, dependent only on $\beta_h$, are:
\begin{align*}
B(\beta_h) &= \frac{1}{t^2 m_H^2} \\
A(\beta_h) &= \frac{\alpha}{t^2 m_A m_H} + \frac{1}{t^2 m_H^2} + \frac{u_0}{t m_H} \\
C(\beta_h) &= \frac{Q-u_0}{t m_A} - \frac{u_0}{t m_H}
\end{align*}
The platform's optimal compensation rate is the vertex of this parabola, $r^* = \frac{A}{2B}$. Substituting this back in yields the platform's optimized value function $V(\beta_h) = \frac{A^2}{4B} + C$. 

Notice that $A(\beta_h)$ can be factored as $A(\beta_h) = \frac{1}{t m_H} \left( \frac{\alpha}{t m_A} + \frac{1}{t m_H} + u_0 \right)$. Dividing $A^2$ by $4B$ perfectly cancels the $\frac{1}{t^2 m_H^2}$ terms, vastly simplifying the value function:
\begin{equation}
V(\beta_h) = \frac{1}{4} \left( \frac{\alpha}{t m_A} + \frac{1}{t m_H} + u_0 \right)^2 + \frac{Q-u_0}{t m_A} - \frac{u_0}{t m_H}
\end{equation}

Now, define the core term of the value function as $W(\beta_h) = \frac{\alpha}{t} y_A + \frac{1}{t} y_H + u_0$. Because $W$ is a positive linear combination of strictly convex functions, $W(\beta_h)$ is strictly convex ($W'' > 0$) and strictly positive ($W > 0$).

We can now rewrite the value function compactly:
\begin{equation}
V(\beta_h) = \frac{1}{4} W(\beta_h)^2 + \frac{Q-u_0}{t} y_A(\beta_h) - \frac{u_0}{t} y_H(\beta_h)
\end{equation}

Taking the total second derivative of $V(\beta_h)$ with respect to $\beta_h$ yields:
\begin{equation}
V''(\beta_h) = \frac{1}{2} (W')^2 + \frac{1}{2} W W'' + \frac{Q-u_0}{t} y_A'' - \frac{u_0}{t} y_H'' \label{eq:V_double_prime}
\end{equation}
We must prove that $V''(\beta_h) > 0$. We already know:
1. $\frac{1}{2} (W')^2 \ge 0$ (a squared real number).
2. Because $Q \ge u_0$ by assumption, and $y_A'' > 0$, the term $\frac{Q-u_0}{t} y_A'' \ge 0$.

Therefore, strict convexity hinges entirely on proving that the remaining terms are strictly positive:
\begin{equation}
\frac{1}{2} W W'' - \frac{u_0}{t} y_H'' > 0
\end{equation}

To sign this final expression, we invoke the human creator's participation constraint: equilibrium effort $q_h^*$ must exceed the outside option $u_0$. 
Recall that $q_h^* = \frac{r^*}{t m_H}$. Using our algebraic definitions from Step 1:
\begin{equation*}
q_h^* = \frac{A / (2B)}{t m_H} = \frac{t^2 m_H^2 A}{2 t m_H} = \frac{1}{2} (t m_H A) = \frac{1}{2} W(\beta_h)
\end{equation*}
Therefore, the participation constraint $q_h^* > u_0$ mandates that $\frac{1}{2}W > u_0$, or simply $W > 2u_0$. 

Substituting this lower bound for $W$ into our critical expression from Step 3:
\begin{align*}
\frac{1}{2} W W'' - \frac{u_0}{t} y_H'' &> u_0 W'' - \frac{u_0}{t} y_H'' \\
&= u_0 \left( \frac{\alpha}{t} y_A'' + \frac{1}{t} y_H'' \right) - \frac{u_0}{t} y_H'' \\
&= \frac{\alpha u_0}{t} y_A''
\end{align*}
Because $\alpha > 0$, $u_0 > 0$, $t > 0$, and $y_A'' > 0$, it strictly follows that $\frac{\alpha u_0}{t} y_A'' \ge 0$.


All constituent parts of Equation \eqref{eq:V_double_prime} are non-negative, and the bounds enforced by the participation constraint ensure the sum is strictly greater than zero. Therefore, $V''(\beta_h) > 0$ for all $\beta_h \in [0,1]$. 

Because the platform's value function is globally strictly convex over a closed interval, it must attain its maximum at one of the extreme boundaries. The optimal policy is a corner solution determined by evaluating $u_p(0)$ and $u_p(1)$. 


\medskip\noindent\textbf{Case (ii): $Q < u_0$.}
When $Q < u_0$, for sufficiently small $\beta_h$ the AI quality $q_A^* = \alpha r^*/(tm_H) + Q$ falls below $u_0$, yielding $D_A = 0$. Since $q_A^*$ is strictly increasing in $\beta_h$ (because $\partial q_A^*/\partial \beta_h = \alpha r^* \delta/(tm_H^2) > 0$), there exists a unique $\tilde\beta \in (0,1)$ satisfying $q_A^*(\tilde\beta) = u_0$. On $[0, \tilde\beta]$ the platform earns only from human demand and strictly benefits from raising $\beta_h$. The relevant domain is therefore $[\tilde\beta, 1]$.
 
On this domain, the envelope derivative is
\[
\frac{du_p}{d\beta_h} = \underbrace{-\frac{(q_A^* - u_0)\,\delta}{t\,m_A^2}}_{\text{friction drag}} + \underbrace{\frac{\alpha r^*\delta}{t^2 m_A m_H^2} + (1-r^*)\!\left(\frac{(q_h^* - u_0)\delta}{t\,m_H^2} + \frac{r^*\delta}{t^2 m_H^3}\right)}_{\equiv\;\Omega(\beta_h)\;>\;0}.
\]
At $\beta_h = \tilde\beta$: the friction drag vanishes (since $q_A^* = u_0$), so $du_p/d\beta_h = \Omega(\tilde\beta) > 0$.
 
At $\beta_h = 1$: $m_A = 1$, $m_H = 1-\delta$, and the friction drag equals $(q_A^*(1) - u_0)\delta/t$. Collecting all positive terms into
\[
\bar\Omega = \frac{\alpha r^*(1)\,\delta}{t^2\,m_H(1)^2} + (1-r^*(1))\!\left(\frac{(q_h^*(1)-u_0)\,\delta}{t\,m_H(1)^2} + \frac{r^*(1)\,\delta}{t^2\,m_H(1)^3}\right),
\]
the derivative at the boundary is $(du_p/d\beta_h)\big|_{\beta_h=1} = -\frac{(q_A^*(1)-u_0)\delta}{t} + \bar\Omega$.
 
Since $du_p/d\beta_h$ is continuous on $(\tilde\beta, 1]$ and positive at $\tilde\beta$:
\begin{itemize}
\item If $\frac{(q_A^*(1)-u_0)\delta}{t} > \bar\Omega$, then $du_p/d\beta_h\big|_1 < 0$, and by the Intermediate Value Theorem there exists $\beta_h^* \in (\tilde\beta, 1)$.
\item Otherwise $du_p/d\beta_h \ge 0$ on the entire interval, giving $\beta_h^* = 1$.
\end{itemize}
This completes the proof.\hfill $\blacksquare$
\end{proof}





\noindent\textbf{Proof of Proposition~\ref{prop:r(eng)}}
\begin{proof}
Substitute the optimal qualities $q_{h}^{*} = \frac{r}{tm_{H}}$ and $q_{A}^{*} = \frac{\alpha r}{tm_{H}} + Q$ into the platform's payoff function:
\begin{equation}
u_{p}(r) = \frac{q_{A}^{2} - u_{0}q_{A}}{tm_{A}} + \frac{q_{h}^{2} - rq_{h} - u_{0}q_{h} + ru_{0}}{tm_{H}}
\end{equation}
Expressing the human net engagement component purely in terms of $r$:
\begin{equation}
(q_{h} - r)D_{h} = \left( \frac{r}{tm_{H}} - r \right) \left( \frac{\frac{r}{tm_{H}} - u_{0}}{tm_{H}} \right) = \frac{r^{2}(1 - tm_{H})}{t^{3}m_{H}^{3}} - \frac{ru_{0}(1 - tm_{H})}{t^{2}m_{H}^{2}}
\end{equation}
Taking the partial derivative with respect to $r$:
\begin{equation}
\frac{\partial ((q_{h} - r)D_{h})}{\partial r} = \frac{2r(1 - tm_{H})}{t^{3}m_{H}^{3}} - \frac{u_{0}(1 - tm_{H})}{t^{2}m_{H}^{2}}
\end{equation}
Next, evaluating the AI engagement component:
\begin{align}
\frac{\partial (q_{A}D_{A})}{\partial r} &= \frac{2q_{A} - u_{0}}{tm_{A}} \frac{\partial q_{A}}{\partial r} = \frac{2\left(\frac{\alpha r}{tm_{H}} + Q\right) - u_{0}}{tm_{A}} \left( \frac{\alpha}{tm_{H}} \right) \nonumber \\
&= \frac{2\alpha^{2}r}{t^{3}m_{H}^{2}m_{A}} + \frac{\alpha(2Q - u_{0})}{t^{2}m_{H}m_{A}}
\end{align}
Summing the derivatives and setting to zero:
\begin{equation}
\frac{2\alpha^{2}r}{t^{3}m_{H}^{2}m_{A}} + \frac{\alpha(2Q - u_{0})}{t^{2}m_{H}m_{A}} + \frac{2r(1 - tm_{H})}{t^{3}m_{H}^{3}} - \frac{u_{0}(1 - tm_{H})}{t^{2}m_{H}^{2}} = 0
\end{equation}
Multiply the entire equation by $\frac{t^{3}m_{H}^{3}m_{A}}{2}$ to clear denominators:
\begin{equation}
\alpha^{2}m_{H}r + \frac{1}{2}\alpha tm_{H}^{2}(2Q - u_{0}) + m_{A}(1 - tm_{H})r - \frac{1}{2}u_{0}tm_{H}m_{A}(1 - tm_{H}) = 0
\end{equation}
Factoring out $r$ yields:
\begin{equation}
r \left[ \alpha^{2}m_{H} + m_{A}(1 - tm_{H}) \right] = \frac{1}{2}u_{0}tm_{H}m_{A}(1 - tm_{H}) - \frac{1}{2}\alpha tm_{H}^{2}(2Q - u_{0})
\end{equation}
Solving for $r$ produces the optimal engagement-based compensation rate $r^{*}$ provided in Proposition \ref{prop:r(eng)}. 
\hfill $\blacksquare$
\end{proof}

\noindent\textbf{Proof of Proposition~\ref{prop:beta(eng)}}
\begin{proof}
By the Envelope Theorem, since $\partial u_p/\partial r = 0$ at $r^*$, the total derivative with respect to $\beta_h$ reduces to the partial derivative evaluated at $r = r^*$:
\[
\frac{du_p}{d\beta_h} = \frac{\partial}{\partial \beta_h}\bigl(q_A^* D_A\bigr) + \frac{\partial}{\partial \beta_h}\bigl((q_h^* - r^*) D_h\bigr).
\]
Holding $r$ fixed, the equilibrium qualities $q_h^* = r/(tm_H)$ and $q_A^* = \alpha r/(tm_H) + Q$ depend on $\beta_h$ through $m_H$ only, with
\[
\frac{\partial q_h^*}{\partial \beta_h} = \frac{r^*\delta}{tm_H^2} > 0, \qquad \frac{\partial q_A^*}{\partial \beta_h} = \frac{\alpha r^*\delta}{tm_H^2} > 0.
\]
 
\medskip\noindent\textbf{AI engagement margin.}
Differentiating $q_A^* D_A = \bigl((q_A^*)^2 - u_0\,q_A^*\bigr)/(tm_A)$:
\[
\Omega_A = \frac{(2q_A^* - u_0)\,\alpha r^*\delta}{t^2 m_A m_H^2} - \frac{q_A^*(q_A^* - u_0)\,\delta}{tm_A^2}.
\]
The first term (quality boost) is positive when $q_A^* > u_0/2$; the second term (friction drag) is positive whenever $q_A^* > u_0$.
 
\medskip\noindent\textbf{Human engagement margin.}
Differentiating $(q_h^* - r^*) D_h = (q_h^* - r^*)(q_h^* - u_0)/(tm_H)$:
\[
\Omega_H = \frac{\delta}{tm_H^2}\!\left[\frac{r^*(2q_h^* - r^* - u_0)}{tm_H} + (q_h^* - r^*)(q_h^* - u_0)\right].
\]
Under Assumption~2(a), $tm_H > 1$, so $q_h^* = r^*/(tm_H) < r^*$ and the factor $(q_h^* - r^*)$ is negative. Consequently, $\Omega_H$ is \emph{not} guaranteed to be positive: the platform may lose net engagement on the human creator's content (paying $r^*$ per view while the quality-weighted engagement per view is only $q_h^* < r^*$). The sign of $\Omega_H$ depends on parameter values and may change across $\beta_h \in [0,1]$.
 
\medskip\noindent\textbf{Total marginal utility.}
The full derivative is
\begin{equation}\label{eq:full-deriv}
\frac{du_p}{d\beta_h} = \Omega_H + \frac{(2q_A^* - u_0)\,\alpha r^*\delta}{t^2 m_A m_H^2} - \frac{q_A^*(q_A^* - u_0)\,\delta}{tm_A^2}.
\end{equation}
 
\medskip\noindent\textbf{Boundary evaluation.}
By Assumption~\ref{ass:eng-promotion}, $du_p/d\beta_h\big|_{\beta_h = 0} > 0$.
 
At $\beta_h = 1$, we have $m_A = 1$ and $m_H = 1-\delta$. All terms in~\eqref{eq:full-deriv} are finite and continuous. The friction drag term $q_A^*(q_A^* - u_0)\delta/t$ is the only term that grows quadratically in $Q$ (through $q_A^* = \alpha r^*/(t(1-\delta)) + Q$), while all other terms grow at most linearly in $Q$. Therefore:
\begin{itemize}
\item For $Q$ sufficiently large, the friction drag dominates and $du_p/d\beta_h\big|_{\beta_h=1} < 0$. Since $du_p/d\beta_h$ is continuous on $[0,1]$, positive at $0$, and negative at $1$, the Intermediate Value Theorem yields an interior $\beta_h^* \in (0,1)$.
\item For $Q$ sufficiently small, the friction drag is weak and $du_p/d\beta_h \geq 0$ on $[0,1]$, giving $\beta_h^* = 1$.
\end{itemize}
\hfill $\blacksquare$
\end{proof}
